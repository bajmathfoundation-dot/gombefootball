      function isValidSeasonCollection() {

        return seasonCollection.matches("Season\\d{4}-\\d{2}");

      }

      

      allow read, write: if isValidSeasonCollection();

      

      match /{teamId} {

    match /GombeTeams/{teamId} {

      allow read, create: if true;

      allow update: if true;

      allow delete: if false;

      

      match /{seasonCollection}/{documentId} {

        allow create: if (

          documentId == '.init' && 

          request.resource.data.keys().hasAll(['initialized', 'note'])

        );

        

        allow delete: if documentId == '.init';

        

        allow read: if seasonCollection.matches("Season\\d{4}-\\d{2}");

        

        // Modified to allow date/time updates without requiring all fields

        allow create, update: if (

          seasonCollection.matches("Season\\d{4}-\\d{2}") && 

          (

            // Allow partial updates for date/time

            request.resource.data.keys().hasAny(['datePlayed', 'time']) ||

            

            // Full document creation requirements

            request.resource.data.keys().hasAll([

              'league_phase', 'matchday', 'datePlayed', 

              'opponent', 'venue'

            ]) || 

            request.resource.data.keys().hasAll([

              'championship', 'matchday', 'datePlayed', 

              'opponent', 'venue'

            ])

          )

        );

        

        allow delete: if seasonCollection.matches("Season\\d{4}-\\d{2}");

      }
